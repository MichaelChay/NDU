// ==UserScript==
// @name         4_S.Listings country fix - ChartNexus
// @namespace    https://ndu.chartnexus.com/tbiz/arts_plus/
// @version      1.0
// @description  Restores incomplete S.Listings market menus in the ARTS+ staging page.
// @author       Fish Head Cracker
// @match        https://tbizfiles.chartnexus.com/cnxideawebv12/widget_*/ncn_stock_listing_panel.jsp*
// @run-at       document-idle
// @sandbox      JavaScript
// @grant        unsafeWindow
// ==/UserScript==

(function () {
    'use strict';

    // Limit this development override to the ARTS+ staging embed.
    const embedUrl = new URLSearchParams(location.search).get('iframeUrl');
    if (!embedUrl) return;
    try {
        const embed = new URL(embedUrl);
        if (embed.origin !== 'https://ndu.chartnexus.com'
            || !embed.pathname.startsWith('/tbiz/arts_plus/')) return;
    } catch (_) {
        return;
    }

    const DEFAULT_MARKET = '2';
    // KLSE, WORLD INDX, SGX, NASDAQ, NYSE, AMEX, IDX, HKSE, SET, PSE.
    // Keep WORLD INDX (11), which was absent from the old fallback list.
    const FALLBACK_MARKETS = ['2', '11', '0', '3', '4', '5', '14', '15', '19', '20'];
    const page = typeof unsafeWindow === 'undefined' ? window : unsafeWindow;
    const observerOptions = { childList: true, subtree: true, characterData: true };
    let observer;
    let repairing = false;

    function parseMarketIds(value) {
        return [...new Set(String(value ?? '').split(',')
            .map(id => id.trim().replace(/^["']|["']$/g, ''))
            .filter(id => /^\d+$/.test(id))
            .map(id => String(Number(id))))];
    }

    function renderedMarketIds(table) {
        return [...table.querySelectorAll('td[id^="mkt_flag_"]')]
            .map(cell => cell.id.slice('mkt_flag_'.length))
            .filter(id => /^\d+$/.test(id));
    }

    function menuNeedsRepair(table, expectedIds) {
        const present = new Set(renderedMarketIds(table));
        // A valid-looking two-row menu can still be incomplete.
        if (expectedIds.some(id => !present.has(id))) return true;

        return [...table.rows].some(row => {
            const label = row.querySelector('[data-i18n]')?.textContent.trim() || '';
            const image = row.querySelector('img')?.getAttribute('src') || '';
            return !image || /(?:^|\/)img\/\.svg(?:\?|$)/.test(image)
                || (page.localeReady === true
                    && (!label || label.includes('cnx-common-exchange-short-')));
        });
    }

    function repairMenu() {
        if (repairing) return false;
        const table = document.querySelector('#mkd_list');
        if (!table || typeof page.fetchMarketList !== 'function'
            || typeof page.chgMarket !== 'function') return false;

        // Preserve additional markets supplied by the page or another mock.
        const expectedIds = [...new Set([
            ...parseMarketIds(page.marketids),
            ...renderedMarketIds(table),
            ...FALLBACK_MARKETS,
        ])];
        if (!menuNeedsRepair(table, expectedIds)) return true;

        const selected = parseMarketIds(page.mkt_selected)[0];
        repairing = true;
        // MutationObserver callbacks are asynchronous: the boolean alone does
        // not prevent the observer from reacting to its own menu rebuild.
        observer?.disconnect();
        try {
            page.marketids = expectedIds.join(',');
            page.fetchMarketList();

            if (!selected || !expectedIds.includes(selected)) {
                page.chgMarket(Number(DEFAULT_MARKET));
            } else {
                // Refresh only the flag. Calling chgMarket would unnecessarily
                // reload stocks and save the selection on every repair.
                const button = document.querySelector('#sel_mkt');
                const flag = document.getElementById('mkt_flag_' + selected);
                if (button && flag) {
                    button.replaceChildren(...[...flag.childNodes].map(node => node.cloneNode(true)));
                }
            }
            document.querySelector('#sel_mkt')?.setAttribute('title', 'Select market');
            console.info('[ChartNexus country fix] Restored ' + expectedIds.length + ' market choices.');
        } catch (error) {
            console.warn('[ChartNexus country fix] Could not rebuild the market menu.', error);
        } finally {
            repairing = false;
            observer?.observe(table, observerOptions);
        }
        return true;
    }

    const startupTimer = setInterval(() => {
        const table = document.querySelector('#mkd_list');
        if (!table || typeof page.fetchMarketList !== 'function'
            || typeof page.chgMarket !== 'function') return;

        clearInterval(startupTimer);
        observer = new MutationObserver(repairMenu);
        observer.observe(table, observerOptions);
        repairMenu();
    }, 100);

    setTimeout(() => clearInterval(startupTimer), 10000);
    window.addEventListener('pagehide', () => {
        clearInterval(startupTimer);
        observer?.disconnect();
    }, { once: true });
})();
