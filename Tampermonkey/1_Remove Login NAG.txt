// ==UserScript==
// @name         1_Remove Login NAG - ChartNexus
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Bypass login overlay and catch cnx_listenSetting instantly
// @author       Fish Head Cracker
// @match        https://ndu.chartnexus.com/tbiz/arts_plus/
// @grant        none
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    console.log("[Tampermonkey] Userscript loaded at document-start!");

    // ==========================================
    // 1. LOGIN OVERLAY BYPASS
    // ==========================================
    function RemoveLoginNAG() {

        console.log("[Tampermonkey] Removing Login");
        document.querySelectorAll('#myLoginModal, .modal-backdrop').forEach(el => el.remove());
        if (document.body) {
            document.body.classList.remove('modal-open');
            document.body.style.overflow = 'auto';
            document.body.style.paddingRight = '0px';
        }
    }

    RemoveLoginNAG();
    const interval = setInterval(RemoveLoginNAG, 500);
    setTimeout(() => clearInterval(interval), 3000);


})();
